schema {
  query: Query
  mutation: Mutation
}

type Query {
  ping: String!
}

type Mutation {
  reserveStock(input: ReserveStockInput!): ReserveStockPayload!
  reserveStockPlain(input: ReserveStockPlainInput!): ReserveStockPlainPayload!
}

type OutOfStock implements Error {
  available: Int!
  message: String!
}

type Reservation {
  sku: String!
  quantity: Int!
}

type ReserveStockPayload {
  reservation: Reservation
  errors: [ReserveStockError!]
}

type ReserveStockPlainPayload {
  reservation: Reservation
  errors: [ReserveStockPlainError!]
}

type SkuInvalid implements Error {
  sku: String!
  message: String!
}

type StockBlocked implements Error {
  sku: String!
  message: String!
}

interface Error {
  message: String!
}

union ReserveStockError = StockBlocked | OutOfStock | SkuInvalid

union ReserveStockPlainError = StockBlocked | OutOfStock | SkuInvalid

input ReserveStockInput {
  sku: String!
  quantity: Int!
}

input ReserveStockPlainInput {
  sku: String!
  quantity: Int!
}

      },
      "errors": null
    }
  }
}
